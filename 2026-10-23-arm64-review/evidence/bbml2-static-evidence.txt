673: int split_kernel_leaf_mapping(unsigned long start, unsigned long end)
674: {
675: 	int ret;
676: 
677: 	/*
678: 	 * If the region is within a pte-mapped area, there is no need to try to
679: 	 * split. Additionally, CONFIG_DEBUG_PAGEALLOC and CONFIG_KFENCE may
680: 	 * change permissions from atomic context so for those cases (which are
681: 	 * always pte-mapped), we must not go any further because taking the
682: 	 * mutex below may sleep. Do not call force_pte_mapping() here because
683: 	 * it could return a confusing result if called from a secondary cpu
684: 	 * prior to finalizing caps. Instead, linear_map_requires_bbml3 gives us
685: 	 * what we need.
686: 	 */
687: 	if (!linear_map_requires_bbml3 || is_kfence_address((void *)start))
688: 		return 0;
689: 
690: 	if (!system_supports_bbml3()) {
691: 		/*
692: 		 * BBML3 systems should not be trying to change
693: 		 * permissions on anything that is not pte-mapped in the first
694: 		 * place. Just return early and let the permission change code
695: 		 * raise a warning if not already pte-mapped.
696: 		 */
697: 		if (system_capabilities_finalized())
698: 			return 0;
699: 
700: 		/*
701: 		 * Boot-time: split_kernel_leaf_mapping_locked() allocates from
702: 		 * page allocator. Can't split until it's available.
703: 		 */
704: 		if (WARN_ON(!page_alloc_available))
705: 			return -EBUSY;
706: 
707: 		/*
708: 		 * Boot-time: Started secondary cpus but don't know if they
709: 		 * support BBML3 yet. Can't allow splitting in this window
710: 		 * in case they don't.
711: 		 */
712: 		if (WARN_ON(num_online_cpus() > 1))
713: 			return -EBUSY;
714: 	}
715: 
716: 	/*
717: 	 * Ensure start and end are at least page-aligned since this is the
718: 	 * finest granularity we can split to.
719: 	 */
720: 	if (start != PAGE_ALIGN(start) || end != PAGE_ALIGN(end))
721: 		return -EINVAL;
722: 
723: 	mutex_lock(&pgtable_split_lock);
724: 	arch_enter_lazy_mmu_mode();
725: 
726: 	/*
727: 	 * The split_kernel_leaf_mapping_locked() may sleep, it is not a
728: 	 * problem for ARM64 since ARM64's lazy MMU implementation allows
729: 	 * sleeping.
730: 	 *
731: 	 * Optimize for the common case of splitting out a single page from a
732: 	 * larger mapping. Here we can just split on the "least aligned" of
733: 	 * start and end and this will guarantee that there must also be a split
734: 	 * on the more aligned address since the both addresses must be in the
735: 	 * same contpte block and it must have been split to ptes.
736: 	 */
737: 	if (end - start == PAGE_SIZE) {
738: 		start = __ffs(start) < __ffs(end) ? start : end;
739: 		ret = split_kernel_leaf_mapping_locked(start);
740: 	} else {
741: 		ret = split_kernel_leaf_mapping_locked(start);
742: 		if (!ret)
743: 			ret = split_kernel_leaf_mapping_locked(end);
744: 	}
745: 
746: 	arch_leave_lazy_mmu_mode();
747: 	mutex_unlock(&pgtable_split_lock);
748: 	return ret;
749: }
